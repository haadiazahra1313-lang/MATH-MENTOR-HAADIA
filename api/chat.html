# api/chat.py import os from flask import Flask, request, jsonify from dotenv import load_dotenv from groq import Groq load_dotenv() app = Flask(__name__) api_key = os.getenv("GROQ_API_KEY") if not api_key: raise RuntimeError( "GROQ_API_KEY was not found. " "Please add it to your environment variables." ) client = Groq( api_key=api_key ) SYSTEM_MESSAGE = { "role": "system", "content": ( "You are Math Mentor, a helpful and intelligent " "AI mathematics assistant. " "Solve mathematics problems accurately. " "Explain solutions clearly and step by step. " "Use simple language when appropriate." ) } @app.route("/api/chat", methods=["POST"]) def chat(): try: data = request.get_json(silent=True) or {} user_input = ( data.get("message") or "" ).strip() if not user_input: return jsonify({ "error": "Message is required." }), 400 messages = [ SYSTEM_MESSAGE, { "role": "user", "content": user_input } ] completion = client.chat.completions.create( model="llama-3.1-8b-instant", messages=messages, temperature=0.7, max_tokens=1024 ) answer = completion.choices[0].message.content return jsonify({ "response": answer }) except Exception as error: return jsonify({ "error": str(error) }), 500 if __name__ == "__main__": app.run( host="0.0.0.0", port=int( os.environ.get( "PORT", 5000 ) ) ) # requirements.txt Flask groq python-dotenv { "version": 2, "builds": [ { "src": "api/chat.py", "use": "@vercel/python" } ], "routes": [ { "src": "/api/chat", "dest": "/api/chat.py" }, { "src": "/chat.html", "dest": "/chat.html" }, { "src": "/", "dest": "/index.html" } ] } # .gitignore .env __pycache__/ *.pyc .venv/ venv/ .vscode/ .idea/
